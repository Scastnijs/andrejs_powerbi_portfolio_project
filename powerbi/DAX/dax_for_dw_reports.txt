Latest Car Density Year = 
MAXX(
    CALCULATETABLE(
        'fact_observation_country',
        REMOVEFILTERS('dim_time'),
        FILTER(
            ALL('dim_indicator'),
            'dim_indicator'[indicator_name] = "car_density"
        )
    ),
    RELATED('dim_time'[year])
)


Latest Car Density = 
VAR LatestYear =
    [Latest Car Density Year]
RETURN
    IF(
        ISBLANK(LatestYear),
        BLANK(),
        CALCULATE(
            MAX('fact_observation_country'[value]),
            REMOVEFILTERS('dim_time'),
            FILTER(
                ALL('dim_indicator'),
                'dim_indicator'[indicator_name] = "car_density"
            ),
            'dim_time'[year] = LatestYear
        )
    )
